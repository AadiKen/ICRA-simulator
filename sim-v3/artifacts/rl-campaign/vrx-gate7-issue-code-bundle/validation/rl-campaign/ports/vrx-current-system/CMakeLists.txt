cmake_minimum_required(VERSION 3.16)
project(vrx_current_relative_velocity)
find_package(gz-sim8 REQUIRED)
add_library(VrxCurrentRelativeVelocity SHARED CurrentRelativeVelocity.cc)
target_link_libraries(VrxCurrentRelativeVelocity PRIVATE gz-sim8::gz-sim8)
set_target_properties(VrxCurrentRelativeVelocity PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
add_library(VrxSurveyorRelativeWind SHARED SurveyorRelativeWind.cc)
target_link_libraries(VrxSurveyorRelativeWind PRIVATE gz-sim8::gz-sim8)
set_target_properties(VrxSurveyorRelativeWind PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
